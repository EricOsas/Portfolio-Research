using System;
using System.IO;
using System.Reflection;
public static class OvernightModelContractTests
{
    static Type model;
    static object Call(string name,params object[] args) {return model.GetMethod(name,BindingFlags.Static|BindingFlags.NonPublic).Invoke(null,args);}
    static void Check(bool value,string name) {if(!value)throw new Exception(name);}
    public static int Main(string[] args)
    {
        AppDomain.CurrentDomain.AssemblyResolve+=(sender,eventArgs)=>
        {
            string dll=new AssemblyName(eventArgs.Name).Name+".dll";
            foreach(string directory in new[]{Path.GetDirectoryName(args[0]),@"C:\Program Files\NinjaTrader 8\bin",@"C:\Users\Administrator\Documents\NinjaTrader 8\bin\Custom"})
            {string path=Path.Combine(directory,dll);if(File.Exists(path))return Assembly.LoadFrom(path);}
            return null;
        };
        model=Assembly.LoadFrom(args[0]).GetType("NinjaTrader.NinjaScript.Strategies.OvernightBiasRangeBreakoutNT8",true);
        Check(!(bool)Call("SignalTime",540.0),"09:00 is ineligible");
        Check((bool)Call("SignalTime",555.0),"09:15 is eligible");
        Check((int)Call("Bias",109.0,100.0,110.0)==1,"upper third long");
        Check((int)Call("Bias",101.0,100.0,110.0)==-1,"lower third short");
        Check((int)Call("Bias",110.0,100.0,110.0)==0,"outside/open boundary abstains");
        Check((int)Call("SignalSide",1,110.0,110.0,100.0,25.0)==0,"boundary close abstains");
        Check((int)Call("SignalSide",1,111.0,110.0,100.0,20.0)==0,"ADX threshold strict");
        Check((int)Call("SignalSide",1,111.0,110.0,100.0,20.01)==1,"directed break");
        Check(double.IsNaN((double)Call("TrailLock",1.499)),"trail activation");
        Check((double)Call("TrailLock",1.5)==0 && (double)Call("TrailLock",2.0)==.5 && (double)Call("TrailLock",2.5)==1,"trail cadence");
        Check((int)Call("Quantity",250.0,251.0)==0 && (int)Call("Quantity",250.0,120.0)==2,"whole-contract risk refusal/round down");
        Check((double)Call("FillRiskCost",14.0,0.0,1,10.0)==14.0,"no fee evidence retains reserved round-trip costs");
        Check((double)Call("FillRiskCost",14.0,15.0,2,10.0)==25.0,"partial entry fees raise round-trip reserve per contract");
        Check(230.0+(double)Call("FillRiskCost",14.0,6.0,1,10.0)>250.0,"observed entry fee breaches formerly affordable 250-dollar fill");
        Check((double)Call("CoveredStop",1,100.0,7.0,20.0,.03,.25)==100.5,"long coverage outward rounding");
        Check((double)Call("CoveredStop",-1,100.0,7.0,20.0,.03,.25)==99.5,"short coverage outward rounding");
        Check((double)Call("BracketPrice",1,120.0,10.0,false)==110 && (double)Call("BracketPrice",1,120.0,10.0,true)==150,"gap quote rebases stop and target");
        Check((double)Call("BracketPrice",-1,80.0,10.0,false)==90 && (double)Call("BracketPrice",-1,80.0,10.0,true)==50,"short next-open geometry");
        string leaseKey="fixture_"+Guid.NewGuid().ToString("N");
        Check((bool)Call("ExposureAcquire",leaseKey,"one"),"first model owns pending exposure");
        Check(!(bool)Call("ExposureAcquire",leaseKey,"two"),"second model cannot race pending exposure");
        Call("ExposureRelease",leaseKey,"two");
        Check(!(bool)Call("ExposureAcquire",leaseKey,"two"),"foreign release refused");
        Call("ExposureRelease",leaseKey,"one");
        Check((bool)Call("ExposureAcquire",leaseKey,"two"),"flat settled owner can release");
        Call("ExposureRelease",leaseKey,"two");
        Check((int)Call("Cutoff",new DateTime(2025,1,9))==715,"preserved special day");
        Check((int)Call("Cutoff",new DateTime(2024,9,3))==870,"normal cutoff");
        // Exercise the actual compiled nested indicator, independent reference recursion.
        Type candle=model.GetNestedType("Candle",BindingFlags.NonPublic), trend=model.GetNestedType("WilderAdx",BindingFlags.NonPublic);
        object adx=Activator.CreateInstance(trend,true);
        double tr=0,up=0,down=0,value=0,prevH=0,prevL=0,prevC=0;
        for(int i=0;i<120;i++)
        {
            double c=100+i*.3+Math.Sin(i*.7)*2,h=c+1,l=c-1;
            object bar=Activator.CreateInstance(candle,new object[]{new DateTime(2022,1,3).AddMinutes(i*15),c,h,l,c});
            trend.GetMethod("Push").Invoke(adx,new[]{bar});
            if(i>0)
            {
                double p=Math.Max(0,h-prevH),n=Math.Max(0,prevL-l);
                if(p==n){p=n=0;}else if(p<n)p=0;else n=0;
                if(i>=15){tr=(tr*14+Math.Max(h-l,Math.Max(Math.Abs(h-prevC),Math.Abs(l-prevC))))/15;up=(up*14+p)/15;down=(down*14+n)/15;}
                double dx=up+down>0?100*Math.Abs(up-down)/(up+down):0;
                if(i>=14)value=(value*14+dx)/15;
            }
            double actual=(double)trend.GetProperty("Value").GetValue(adx,null);
            Check(Math.Abs(actual-value)<1e-10,"MT5 Wilder recursion at "+i);
            prevH=h;prevL=l;prevC=c;
        }
        Console.WriteLine("PASS Overnight actual-source timing, bias, breakout, cadence, calendar and 120-bar MT5 ADX contracts");
        return 0;
    }
}
